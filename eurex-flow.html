withVwap = rows => {
  let pq = 0, q = 0, day = null;
  return rows.map((r, i) => {
    const d = r.t.slice(0, 10);
    if (d !== day) { pq = 0; q = 0; day = d; }
    pq += r.v * r.q; q += r.q;
    return {...r, i, day: d, time: r.t.slice(11, 16), vwap: pq / q, prev: i ? rows[i - 1].v : r.v,
      first: i === 0 || rows[i - 1].t.slice(0, 10) !== d};
  });
}

// Outliers are left out of the charts and the VWAP but kept, flagged, in the trade table.
// Stage 1 (gross misprints): more than `k` median absolute deviations, and at least `floor`, from the series median.
grossSplit = (rows, floor, k = 8) => {
  const med = d3.median(rows, r => r.v);
  const mad = d3.median(rows, r => Math.abs(r.v - med)) || 0;
  const limit = Math.max(floor, k * mad);
  return {kept: rows.filter(r => Math.abs(r.v - med) <= limit), dropped: rows.filter(r => Math.abs(r.v - med) > limit)};
}
// Stage 2 (off-market prints): more than `k` typical deviations, and at least `floor`, from that day's median.
localSplit = (rows, floor, k = 8) => {
  const dayMedian = new Map(d3.rollups(rows, g => d3.median(g, r => r.v), r => r.t.slice(0, 10)));
  const dev = r => Math.abs(r.v - dayMedian.get(r.t.slice(0, 10)));
  const limit = Math.max(floor, k * (d3.median(rows, dev) || 0));
  return {kept: rows.filter(r => dev(r) <= limit), dropped: rows.filter(r => dev(r) > limit)};
}
withOutliers = (rows, outliers) => [...rows, ...outliers.map(r => ({...r, outlier: true, time: r.t.slice(11, 16)}))]

// Stats for the latest trading day in the series.
tilesFor = (rows, unit, digits) => {
  if (!rows.length) return html`<p class="text-muted">No trades in this series.</p>`;
  const r = rows[rows.length - 1], today = rows.filter(d => d.day === r.day);
  return html`<div class="tiles">
    ${tile("Latest day", r.day, `${fmt(today.length)} trades`)}
    ${tile("Lots", fmt(d3.sum(today, d => d.q)))}
    ${tile("VWAP", `${r.vwap.toFixed(digits)} ${unit}`)}
    ${tile("Last", `${r.v.toFixed(digits)} ${unit}`, r.time)}
    ${tile("Day range", `${d3.min(today, d => d.v).toFixed(digits)} – ${d3.max(today, d => d.v).toFixed(digits)}`)}
  </div>`;
}

// floating legs (the trade file has no index field). Split them with a 1-D two-means so each VWAP is like-for-like.
twoMeans = values => {
  let lo = d3.quantile(values, 0.1), hi = d3.quantile(values, 0.9);
  for (let k = 0; k < 20; k++) {
    const mid = (lo + hi) / 2;
    lo = d3.mean(values.filter(v => v <= mid)) ?? lo;
    hi = d3.mean(values.filter(v => v > mid)) ?? hi;
  }
  return {lo, hi, cut: (lo + hi) / 2, split: hi - lo >= 8};
}

NoteData and method
  • EFP and EFS trades: Eurex EFP-Fin and EFS trade files. Eurex lists only about 20 trading days, so every file is archived locally and the data refreshed hourly. Only the front contract (December 2026) is shown.
  • CTD and 2nd CTD: deliverables ranked by price / conversion factor at today’s yields, as on the CTD page. EFPs against any other bond are excluded.
  • Gross basis = cash price − futures × CF (cents). It includes carry to delivery, so it drifts down towards the net basis as delivery approaches.
  • Invoice spread = swap fixed rate − the CTD yield implied by the futures price, treating futures × CF as the CTD’s forward price on the delivery date. Swap tenors in an EFS vary, so compare spreads across trades with similar tenors.
  • Monthly open interest: Eurex monthly statistics, with yields from the Bundesbank (Germany, month-end) and FRED (France and Italy, monthly averages).
